---
- name: Install Nginx
  ansible.builtin.package:
    name: nginx
    state: present

- name: Ensure certmonger is running
  ansible.builtin.service:
    name: certmonger
    state: started
    enabled: true
- name: Prepare SAN config file
  ansible.builtin.shell: |
    cat /etc/ssl/openssl.cnf <(printf "\n[SAN]\nsubjectAltName=DNS:{{ inventory_hostname.split('.')[0] }}.{{ ipa_domain }},IP:{{ hostvars['helper']['ansible_host'] }}") > /tmp/openssl-san.cnf
  args:
    executable: /bin/bash
    creates: /tmp/openssl-san.cnf

- name: Generate Private Key
  ansible.builtin.command:
    cmd: "openssl genrsa -out /tmp/nginx.key 4096"
    creates: /tmp/nginx.key

- name: Generate CSR with SAN
  ansible.builtin.command:
    cmd: >
      openssl req -new 
      -key /tmp/nginx.key 
      -out /tmp/nginx-san.csr 
      -subj "/CN={{ inventory_hostname.split('.')[0] }}.{{ ipa_domain }}" 
      -reqexts SAN 
      -config /tmp/openssl-san.cnf
    creates: /tmp/nginx-san.csr

- name: make sure you login
  become: true
  command: "kinit -k host/{{ inventory_hostname.split('.')[0] }}.{{ ipa_domain }}"

- name: Request IPA Certificate (CLI Method)
  become: true
  ansible.builtin.shell: |
    set -o pipefail
    ipa cert-request /tmp/nginx-san.csr \
      --principal=host/{{ inventory_hostname.split('.')[0] }}.{{ ipa_domain }} | \
      grep "Serial number"|grep -v hex | awk '{print $NF}'
  register: cert_serno
  changed_when: cert_serno.rc == 0

- name: Fetch Signed Certificate
  ansible.builtin.command:
    cmd: "ipa cert-show {{ cert_serno.stdout | trim | int}} --out /tmp/nginx.crt"
    creates: /tmp/nginx.crt

- name: Verify Modulus Match (Key vs Cert)
  ansible.builtin.shell: |
    set -o pipefail
    KMOD=$(openssl rsa -noout -modulus -in /tmp/nginx.key | openssl sha256)
    CMOD=$(openssl x509 -noout -modulus -in /tmp/nginx.crt | openssl sha256)
    [ "$KMOD" = "$CMOD" ]
  changed_when: false
  register: keymatch

- name: Move and Secure Certificates
  become: true
  block:
    - name: Ensure cert directory exists
      ansible.builtin.file:
        path: /etc/nginx/certs
        state: directory
        owner: root
        group: root
        mode: '0755'

    - name: Copy IPA Root CA
      ansible.builtin.copy:
        src: /etc/ipa/ca.crt
        dest: /etc/nginx/certs/root-ca.crt
        remote_src: true
        mode: '0644'

    - name: Deploy Key and Cert
      ansible.builtin.copy:
        src: "/tmp/{{ item.src }}"
        dest: "/etc/nginx/certs/{{ item.dest }}"
        remote_src: true
        mode: "{{ item.mode }}"
      loop:
        - { src: 'nginx.key', dest: 'nginx.key', mode: '0600' }
        - { src: 'nginx.crt', dest: 'nginx.crt', mode: '0644' }

    - name: Build Fullchain
      ansible.builtin.shell: "cat /etc/nginx/certs/nginx.crt /etc/nginx/certs/root-ca.crt > /etc/nginx/certs/nginx-fullchain.crt"
  when: keymatch.rc == 0

- name: Install minimal nginx.conf
  run_once: true
  template:
    src: nginx.conf.j2
    dest: /etc/nginx/nginx.conf
    mode: '0644'
    force: true
  notify: restart nginx

- name: SELinux and Nginx Config
  become: true
  block:
    - name: Apply SELinux context
      ansible.builtin.command: "restorecon -Rv /etc/nginx/certs"
      changed_when: false

    - name: Install Cloud Helper Config
      ansible.builtin.copy:
        dest: "/etc/nginx/conf.d/cloud_helper.conf"
        mode: "0644"
        content: |
          server {
              listen {{ nat_ip_device }}:443 ssl;
              server_name {{ inventory_hostname.split('.')[0] }}.{{ ipa_domain }};
              
              ssl_certificate     /etc/nginx/certs/nginx-fullchain.crt;
              ssl_certificate_key /etc/nginx/certs/nginx.key;

              location /ignition/ { alias {{ cloud_helper_workdir }}/ignition/; }
              location /fcos/     { alias {{ cloud_helper_workdir }}/fcos/; }
          }
          server {
              listen {{ nat_ip_device }}:80;
              server_name {{ inventory_hostname.split('.')[0] }}.{{ ipa_domain }};
              
              ssl_certificate     /etc/nginx/certs/nginx-fullchain.crt;
              ssl_certificate_key /etc/nginx/certs/nginx.key;

              location /ignition/ { alias {{ cloud_helper_workdir }}/ignition/; }
              location /fcos/     { alias {{ cloud_helper_workdir }}/fcos/; }
          }

      notify: restart nginx
  when: keymatch.rc == 0
  notify: restart nginx

- name: Disable ALL listeners in nginx.conf.default
  replace:
    path: /etc/nginx/nginx.conf.default
    regexp: '^\s*listen\s+.*;'
    replace: '# listen disabled by helper_node_prepare'
  notify: restart nginx

- name: Ensure xning is running
  ansible.builtin.service:
    name: nginx
    state: restarted
    enabled: true
